# Needlbar v0.3 System Monitor Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make Needlbar a configurable Stats replacement that shows system
metrics and AI usage in one adaptive menu-bar item and complete popover.

**Architecture:** Keep macOS collection and presentation in Swift, retain the
existing Rust provider bridge, and merge both streams in a Swift
`CombinedSnapshotStore`. A protocol-backed system metrics service emits a
one-second system snapshot; public IP is an opt-in, cached HTTPS dependency.
The existing AppKit panel presenter remains the single anchoring and dismissal
boundary.

**Tech Stack:** Swift 6, SwiftUI, AppKit, Foundation, IOKit, Swift Testing,
existing `NeedlbarCore`/`NeedlbarApp` targets, and the existing `make test` gate.

public struct MetricPercentage: Equatable, Sendable {
    public let value: Double

    public init?(_ value: Double?) {
        guard let value, value.isFinite, (0...100).contains(value) else { return nil }
        self.value = value
    }
}

public enum MetricAvailability: Equatable, Sendable {
    case fresh(capturedAt: Date)
    case stale(lastSuccessfulAt: Date)
    case unavailable(code: String)
}

public enum MonitorModuleID: String, CaseIterable, Codable, Sendable {
    case cpu, memory, disk, network, battery, ai

    public static let defaultOrder: [MonitorModuleID] = [.cpu, .memory, .disk, .network, .battery, .ai]
}

public enum AIProviderDisplayMetric: String, CaseIterable, Codable, Sendable {
    case usage, remaining, cost, connectionStatus
}

public struct AIProviderDisplayPreference: Codable, Equatable, Sendable {
    public var isVisible: Bool
    public var metric: AIProviderDisplayMetric

    public init(isVisible: Bool = true, metric: AIProviderDisplayMetric = .usage) {
        self.isVisible = isVisible
        self.metric = metric
    }
}

public struct SystemMetricsSnapshot: Equatable, Sendable {
    public struct CPU: Equatable, Sendable {
        public let totalUsage: MetricPercentage?
        public let perCoreUsage: [MetricPercentage]
    }

    public struct Memory: Equatable, Sendable {
        public let usedBytes: UInt64?
        public let freeBytes: UInt64?
        public let swapUsedBytes: UInt64?
        public let pressure: String?
    }

    public struct DiskVolume: Equatable, Sendable {
        public let name: String
        public let usedBytes: UInt64?
        public let freeBytes: UInt64?
        public let readBytesPerSecond: UInt64?
        public let writeBytesPerSecond: UInt64?
    }

    public struct Network: Equatable, Sendable {
        public let uploadBytesPerSecond: UInt64?
        public let downloadBytesPerSecond: UInt64?
        public let localIPAddresses: [String]
        public let publicIPAddress: String?
    }

    public struct Battery: Equatable, Sendable {
        public let level: MetricPercentage?
        public let isCharging: Bool?
        public let health: MetricPercentage?
    }

    public let capturedAt: Date
    public let cpu: CPU
    public let memory: Memory
    public let disks: [DiskVolume]
    public let network: Network
    public let battery: Battery
    public let availability: [MonitorModuleID: MetricAvailability]

    public init(
        capturedAt: Date,
        cpu: CPU,
        memory: Memory,
        disks: [DiskVolume],
        network: Network,
        battery: Battery,
        availability: [MonitorModuleID: MetricAvailability]

        self.capturedAt = capturedAt
        self.cpu = cpu
        self.memory = memory
        self.disks = disks
        self.network = network
        self.battery = battery
        self.availability = availability
    }
}

public struct SystemMonitorConfiguration: Codable, Equatable, Sendable {
    public var order: [MonitorModuleID]
    public var visibleModules: Set<MonitorModuleID>
    public var publicIPEnabled: Bool
    public var ai: [ProviderID: AIProviderDisplayPreference]

    public init(
        order: [MonitorModuleID] = MonitorModuleID.defaultOrder,
        visibleModules: Set<MonitorModuleID> = Set(MonitorModuleID.defaultOrder),
        publicIPEnabled: Bool = false,
        ai: [ProviderID: AIProviderDisplayPreference] = Dictionary(
            uniqueKeysWithValues: ProviderID.allCases.map { ($0, AIProviderDisplayPreference()) }
        )
    ) {
        self.order = order
        self.visibleModules = visibleModules
        self.publicIPEnabled = publicIPEnabled
        self.ai = ai
    }
}

- [ ] **Step 4: Implement the service.** `SystemMetricsService` is an actor
  with `start(publicIPEnabled:)`, `stop()`, `updates()`, and an internal
  `tickForTesting()`. It publishes the latest good metric value with
  `.fresh`, retains the previous value as `.stale` after a failed tick, and
  emits `.unavailable(code:)` when no successful value exists. The one-second
  loop only invokes the system collector. Public IP is requested on the first
  enabled tick and again after 300 seconds, bounded by a two-second timeout;
  DNS, TLS, HTTP, parse, and timeout errors become `publicIPUnavailable`.

- [ ] **Step 5: Implement macOS collectors without shell commands.** In

  - use `host_statistics64` for CPU and memory counters;
  - use `IOPSCopyPowerSourcesInfo`/`IOPSCopyPowerSourcesList` for battery;
  - use `FileManager` resource values for mounted-volume capacity;
  - use `getifaddrs` and `if_data` byte counters for local interfaces and
    transfer rates; and
  - normalize interface addresses and discard loopback/link-local values from
    the default local-IP list.

  No collector may log raw addresses, paths, or API error payloads.

```swift
@MainActor
final class SystemMonitorSettingsModel: ObservableObject {
    private let configuration: ModuleConfiguration

    init(configuration: ModuleConfiguration) {
        self.configuration = configuration
    }

    func setVisible(_ module: MonitorModuleID, _ visible: Bool) {
        var value = configuration.systemMonitor
        if visible { value.visibleModules.insert(module) }
        else { value.visibleModules.remove(module) }
        configuration.setSystemMonitor(value)
    }

    func move(_ module: MonitorModuleID, before target: MonitorModuleID) {
        var value = configuration.systemMonitor
        value.order.removeAll { $0 == module }
        guard let index = value.order.firstIndex(of: target) else { return }
        value.order.insert(module, at: index)
        configuration.setSystemMonitor(value)
    }

    func setPublicIPEnabled(_ enabled: Bool) {
        var value = configuration.systemMonitor
        value.publicIPEnabled = enabled
        configuration.setSystemMonitor(value)
    }

    func setAIProvider(_ provider: ProviderID, metric: AIProviderDisplayMetric) {
        var value = configuration.systemMonitor
        var preference = value.ai[provider] ?? AIProviderDisplayPreference()
        preference.metric = metric
        value.ai[provider] = preference
        configuration.setSystemMonitor(value)
    }
}
```
